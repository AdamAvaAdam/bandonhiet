CẤU HÌNH MÀU SẮC THEO CẤP ĐỘ (GAME STYLE) function getRankColor(ds) { if (ds <= 0) return '#b2bec3'; // Xám (Trống) if (ds < 100000000) return '#00b894'; // Xanh nhạt (Uncommon) if (ds < 500000000) return '#a29bfe'; // Tím (Epic) if (ds < 1500000000) return '#fdcb6e'; // Cam (Legendary) return '#ff1744'; // Đỏ (Mythic) } // 3. HÀM LOAD KHO CỐ ĐỊNH (Gọi trong enterApp) function loadCompanyWarehouses() { const defaultRadius = parseFloat(document.getElementById('radiusInput').value) || 3; COMPANY_WAREHOUSES.forEach(w => { addWarehouse(L.latLng(w.lat, w.lng), defaultRadius, w.name, true); }); } // 4. CẬP NHẬT HÀM addWarehouse ĐỂ CÓ TAG TÊN function addWarehouse(latlng, radiusKm, name, isCompany = false) { const id = Date.now() + Math.random(); const rM = radiusKm * 1000; // Tạo Marker với Label vĩnh viễn (Permanent Tooltip) const m = L.marker(latlng, { draggable: true, icon: L.divIcon({ className: 'custom-div-icon', html: `
${name}
`, iconAnchor: [15, 0] }) }).addTo(map); const c = L.circle(latlng, { radius: rM, color: isCompany ? '#2d3436' : '#ff1744', weight: 2, fillOpacity: 0.05, interactive: false }).addTo(map); const wh = { id, marker: m, circle: c, latlng, radius: rM, name, isCompany }; m.on('dragend', () => { wh.latlng = m.getLatLng(); c.setLatLng(wh.latlng); updateStats(); saveWarehouses(); }); // Cho phép đổi bán kính riêng bằng cách click vào kho m.on('dblclick', () => { const newR = prompt("Nhập bán kính mới cho " + name + " (km):", wh.radius/1000); if(newR) { wh.radius = parseFloat(newR) * 1000; c.setRadius(wh.radius); updateStats(); saveWarehouses(); } }); warehouses.push(wh); updateStats(); if(!isCompany) saveWarehouses(); }
